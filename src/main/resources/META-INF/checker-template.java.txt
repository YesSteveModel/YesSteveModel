        private static boolean methodExists(
                Class<?> owner,
                String name,
                Class<?> returnType,
                Class<?>[] parameters,
                boolean staticMember,
                boolean searchParents,
                java.util.Set<Class<?>> visited) {
            if (owner == null || !visited.add(owner)) {
                return false;
            }
            for (java.lang.reflect.Method method : owner.getDeclaredMethods()) {
                if (method.getName().equals(name)
                        && method.getReturnType() == returnType
                        && java.util.Arrays.equals(method.getParameterTypes(), parameters)
                        && java.lang.reflect.Modifier.isStatic(method.getModifiers()) == staticMember) {
                    return true;
                }
            }
            if (!searchParents) {
                return false;
            }
            if (methodExists(owner.getSuperclass(), name, returnType, parameters,
                    staticMember, true, visited)) {
                return true;
            }
            for (Class<?> interfaceType : owner.getInterfaces()) {
                if (methodExists(interfaceType, name, returnType, parameters,
                        staticMember, true, visited)) {
                    return true;
                }
            }
            return false;
        }

        private static java.lang.reflect.Field findField(
                Class<?> owner,
                String name,
                java.util.Set<Class<?>> visited) {
            if (owner == null || !visited.add(owner)) {
                return null;
            }
            try {
                return owner.getDeclaredField(name);
            } catch (NoSuchFieldException ignored) {
                java.lang.reflect.Field inherited = findField(owner.getSuperclass(), name, visited);
                if (inherited != null) {
                    return inherited;
                }
                for (Class<?> interfaceType : owner.getInterfaces()) {
                    inherited = findField(interfaceType, name, visited);
                    if (inherited != null) {
                        return inherited;
                    }
                }
                return null;
            }
        }

        private static Class<?> resolveType(String token)
                throws ClassNotFoundException, ProbeFailure {
            int dimensions = 0;
            while (dimensions < token.length() && token.charAt(dimensions) == '[') {
                dimensions++;
            }
            if (dimensions == token.length()) {
                throw new ProbeFailure("Invalid type token: " + token);
            }
            String baseToken = token.substring(dimensions);
            Class<?> base = switch (baseToken.charAt(0)) {
                case 'V' -> void.class;
                case 'Z' -> boolean.class;
                case 'B' -> byte.class;
                case 'S' -> short.class;
                case 'I' -> int.class;
                case 'J' -> long.class;
                case 'C' -> char.class;
                case 'F' -> float.class;
                case 'D' -> double.class;
                case 'L' -> Class.forName(baseToken.substring(1), false, loader());
                case 'P' -> resolveProbeType(baseToken.substring(1));
                default -> throw new ProbeFailure("Unknown type token: " + token);
            };
            for (int index = 0; index < dimensions; index++) {
                base = java.lang.reflect.Array.newInstance(base, 0).getClass();
            }
            return base;
        }

        private static Class<?> resolveProbeType(String probeToken) throws ProbeFailure {
            int separator = probeToken.lastIndexOf('#');
            if (separator <= 0 || separator == probeToken.length() - 1) {
                throw new ProbeFailure("Invalid reobf type probe token: " + probeToken);
            }
            String className = probeToken.substring(0, separator);
            String fieldName = probeToken.substring(separator + 1);
            try {
                return Class.forName(className, false, loader())
                        .getDeclaredField(fieldName).getType();
            } catch (Throwable throwable) {
                throw new ProbeFailure(describe(throwable));
            }
        }

        private static String describe(Throwable throwable) {
            Throwable current = throwable;
            if (throwable instanceof java.lang.reflect.InvocationTargetException invocation

        }
